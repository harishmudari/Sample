// Example 8: Error Handling and Recovery
public class RobustTest {
    
    @Test
    public void testWithErrorRecovery() {
        EnterpriseSeleniumKeywords.initializeChromeDriver(false, false, false, true);
        
        try {
            // This might fail due to network issues
            boolean navigated = EnterpriseSeleniumKeywords.navigateToUrl("https://unreliable-site.com");
            
            if (!navigated) {
                // Fallback to backup URL
                navigated = EnterpriseSeleniumKeywords.navigateToUrl("https://backup-site.com");
                Assert.assertTrue("Both primary and backup sites failed", navigated);
            }
            
            // Try to click an element that might not exist
            boolean clicked = EnterpriseSeleniumKeywords.clickElementSmart(By.id("submitBtn"));
            
            if (!clicked) {
                // Try alternative locator
                clicked = EnterpriseSeleniumKeywords.clickElementSmart(By.name("submit"));
                Assert.assertTrue("Could not find submit button with any locator", clicked);
            }
            
        } catch (EnterpriseSeleniumKeywords.CustomElementNotFoundException e) {
            // Handle custom exceptions
            System.out.println("Element not found: " + e.getMessage());
            // Take screenshot for debugging
            EnterpriseSeleniumKeywords.takeScreenshotOnFailure("testWithErrorRecovery");
            throw e;
        }
    }
    
    @AfterMethod
    public void tearDown() {
        EnterpriseSeleniumKeywords.quitDriver();
    }
}
