import org.openqa.selenium.WebDriver;
import org.openqa.selenium.chrome.ChromeDriver;
import org.openqa.selenium.remote.RemoteWebDriver;
import org.openqa.selenium.remote.SessionId;

import java.io.FileWriter;
import java.io.IOException;

public class StartSession {
    public static void main(String[] args) throws IOException {
        // Start new Chrome browser session
        WebDriver driver = new ChromeDriver();

        // Get session ID and executor URL
        SessionId sessionId = ((RemoteWebDriver) driver).getSessionId();
        String executorURL = ((RemoteWebDriver) driver).getCommandExecutor().getAddressOfRemoteServer().toString();

        System.out.println("Session ID: " + sessionId);
        System.out.println("Executor URL: " + executorURL);

        // Save session details to file
        try (FileWriter writer = new FileWriter("session.txt")) {
            writer.write(sessionId.toString() + "\n");
            writer.write(executorURL);
        }

        // Keep the browser open for reattachment
        driver.get("https://example.com");
    }
}

import org.apache.poi.ss.usermodel.*;
import org.apache.poi.xssf.usermodel.XSSFWorkbook;
import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;
import org.openqa.selenium.remote.HttpCommandExecutor;
import org.openqa.selenium.remote.RemoteWebDriver;
import org.openqa.selenium.remote.SessionId;

import java.io.*;
import java.net.URL;

public class ReconnectAndResumeTest {
    public static void main(String[] args) throws Exception {
        WebDriver driver;
        String sessionIdStr = null;
        String executorURL = null;

        // Read session.txt
        try (BufferedReader reader = new BufferedReader(new FileReader("session.txt"))) {
            sessionIdStr = reader.readLine();
            executorURL = reader.readLine();
        }

        try {
            HttpCommandExecutor executor = new HttpCommandExecutor(new URL(executorURL));
            RemoteWebDriver remoteDriver = new RemoteWebDriver(executor, new org.openqa.selenium.remote.DesiredCapabilities());
            remoteDriver.setSessionId(sessionIdStr);
            driver = remoteDriver;
        } catch (Exception e) {
            System.out.println("Unable to reconnect. Starting new session.");
            driver = new ChromeDriver();
        }

        // Load Excel
        FileInputStream fis = new FileInputStream("testSteps.xlsx");
        Workbook workbook = new XSSFWorkbook(fis);
        Sheet sheet = workbook.getSheetAt(0);

        int lastPassedStep = 0;

        // Identify last executed step with PASS and Execute = Yes
        for (Row row : sheet) {
            if (row.getRowNum() == 0) continue; // skip header
            Cell execCell = row.getCell(5);
            Cell statusCell = row.getCell(6);
            if ("Yes".equalsIgnoreCase(execCell.getStringCellValue()) && "PASS".equalsIgnoreCase(statusCell.getStringCellValue())) {
                lastPassedStep = (int) row.getCell(0).getNumericCellValue();
            }
        }

        // Execute from next step
        for (Row row : sheet) {
            if (row.getRowNum() == 0) continue;
            int stepNum = (int) row.getCell(0).getNumericCellValue();
            if (stepNum <= lastPassedStep) continue;

            String execute = row.getCell(5).getStringCellValue();
            if (!"Yes".equalsIgnoreCase(execute)) {
                row.createCell(6).setCellValue("SKIP");
                continue;
            }

            String keyword = row.getCell(4).getStringCellValue();
            String xpath = row.getCell(2).getStringCellValue();
            String data = row.getCell(3).getStringCellValue();

            try {
                switch (keyword.toLowerCase()) {
                    case "click":
                        WebElement btn = driver.findElement(By.xpath(xpath));
                        btn.click();
                        break;
                    case "input":
                        WebElement field = driver.findElement(By.xpath(xpath));
                        field.clear();
                        field.sendKeys(data);
                        break;
                    case "verify":
                        driver.findElement(By.xpath(xpath)); // simple presence check
                        break;
                    case "open":
                        driver.get(data);
                        break;
                    default:
                        throw new Exception("Unknown keyword: " + keyword);
                }
                row.createCell(6).setCellValue("PASS");
            } catch (Exception e) {
                row.createCell(6).setCellValue("FAIL");
                break;
            }
        }

        fis.close();

        // Save results back to Excel
        FileOutputStream fos = new FileOutputStream("testSteps.xlsx");
        workbook.write(fos);
        workbook.close();
        fos.close();

        driver.quit();
    }
}

import org.apache.poi.ss.usermodel.*;
import org.apache.poi.xssf.usermodel.XSSFWorkbook;

    public static void run(String excelFilePath) throws Exception {
        WebDriver driver;
        String sessionIdStr = null;
        String executorURL = null;

        ExtentReports extent = ReportManager.getInstance();
        ExtentTest test = extent.createTest(new File(excelFilePath).getName());

        // Attempt session reconnect if available
        try (BufferedReader reader = new BufferedReader(new FileReader("session.txt"))) {
            sessionIdStr = reader.readLine();
            executorURL = reader.readLine();
        } catch (IOException e) {
            test.warning("No previous session found. Starting fresh.");
        }

        try {
            if (sessionIdStr != null && executorURL != null) {
                HttpCommandExecutor executor = new HttpCommandExecutor(new URL(executorURL));
                RemoteWebDriver remoteDriver = new RemoteWebDriver(executor, new org.openqa.selenium.remote.DesiredCapabilities());
                remoteDriver.setSessionId(sessionIdStr);
                driver = remoteDriver;
                test.info("Reconnected to existing session");
            } else {
                driver = new ChromeDriver();
                test.info("Started new ChromeDriver session");
            }
        } catch (Exception e) {
            test.warning("Unable to reconnect. Starting new session.");
            driver = new ChromeDriver();
        }

        FileInputStream fis = new FileInputStream(excelFilePath);
        Workbook workbook = new XSSFWorkbook(fis);
        Sheet sheet = workbook.getSheetAt(0);

        int lastPassedStep = 0;

        for (Row row : sheet) {
            if (row.getRowNum() == 0) continue;
            if ("Yes".equalsIgnoreCase(row.getCell(5).getStringCellValue()) &&
                "PASS".equalsIgnoreCase(row.getCell(6).getStringCellValue())) {
                lastPassedStep = (int) row.getCell(0).getNumericCellValue();
            }
        }

        for (Row row : sheet) {
            if (row.getRowNum() == 0) continue;
            int stepNum = (int) row.getCell(0).getNumericCellValue();
            if (stepNum <= lastPassedStep) continue;

            String execute = row.getCell(5).getStringCellValue();
            if (!"Yes".equalsIgnoreCase(execute)) {
                row.createCell(6).setCellValue("SKIP");
                test.skip("Step " + stepNum + " skipped as per Execute = No");
                continue;
            }

            String description = row.getCell(1).getStringCellValue();
            String keyword = row.getCell(4).getStringCellValue();
            String xpath = row.getCell(2).getStringCellValue();
            String data = row.getCell(3).getStringCellValue();

            try {
                switch (keyword.toLowerCase()) {
                    case "click":
                        WebElement btn = driver.findElement(By.xpath(xpath));
                        btn.click();
                        break;
                    case "input":
                        WebElement field = driver.findElement(By.xpath(xpath));
                        field.clear();
                        field.sendKeys(data);
                        break;
                    case "verify":
                        driver.findElement(By.xpath(xpath));
                        break;
                    case "open":
                        driver.get(data);
                        break;
                    default:
                        throw new Exception("Unknown keyword: " + keyword);
                }
                row.createCell(6).setCellValue("PASS");
                test.pass("Step " + stepNum + ": " + description);
            } catch (Exception e) {
                row.createCell(6).setCellValue("FAIL");
                String screenshotPath = takeScreenshot(driver, "Step_" + stepNum);
                test.fail("Step " + stepNum + ": " + description + " FAILED",
                        MediaEntityBuilder.createScreenCaptureFromPath(screenshotPath).build());
                break;
            }
        }

        fis.close();
        FileOutputStream fos = new FileOutputStream(excelFilePath);
        workbook.write(fos);
        workbook.close();
        fos.close();

        extent.flush();
        driver.quit();
    }

    private static String takeScreenshot(WebDriver driver, String name) {
        try {
            TakesScreenshot ts = (TakesScreenshot) driver;
            File src = ts.getScreenshotAs(OutputType.FILE);
            String timestamp = new SimpleDateFormat("yyyyMMdd_HHmmss").format(new Date());
            String path = "reports/screenshots/" + name + "_" + timestamp + ".png";
            File dest = new File(path);
            dest.getParentFile().mkdirs();
            org.openqa.selenium.io.FileHandler.copy(src, dest);
            return path;
        } catch (Exception e) {
            return null;
        }
    }
}
