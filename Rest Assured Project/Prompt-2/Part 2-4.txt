package com.harish.api.framework.config;

import org.apache.logging.log4j.LogManager;
import org.apache.logging.log4j.Logger;

import java.io.IOException;
import java.io.InputStream;
import java.util.Properties;

/**
 * FrameworkConfig is responsible for loading and providing configuration properties
 * from environment-specific property files.
 * 
 * <p>This class reads properties from:
 * <ul>
 *   <li>application.properties (base configuration)</li>
 *   <li>application-{env}.properties (environment-specific overrides)</li>
 * </ul>
 * 
 * <p>The environment is determined by the system property "env" (default: DEV).
 * 
 * <p>Usage: This class is used internally by EnvironmentManager and should not be
 * instantiated directly by test classes.
 * 
 * @author Harish
 * @version 1.0
 */
public class FrameworkConfig {

    private static final Logger logger = LogManager.getLogger(FrameworkConfig.class);
    
    private final Properties properties;
    private final String environment;

    // Property file paths
    private static final String BASE_CONFIG_FILE = "config/application.properties";
    private static final String ENV_CONFIG_FILE_TEMPLATE = "config/application-%s.properties";

    /**
     * Constructor that loads configuration based on the environment.
     * 
     * <p>Environment is determined by system property "env" (DEV/QA/STAGING/PROD).
     * Falls back to "DEV" if not specified.
     */
    public FrameworkConfig() {
        this.environment = System.getProperty("env", "DEV").toUpperCase();
        this.properties = new Properties();
        loadConfiguration();
    }

    /**
     * Loads configuration from base and environment-specific property files.
     * 
     * <p>Loading order:
     * <ol>
     *   <li>Base properties from application.properties</li>
     *   <li>Environment-specific properties from application-{env}.properties (overrides base)</li>
     * </ol>
     */
    private void loadConfiguration() {
        try {
            // Load base configuration
            loadPropertiesFile(BASE_CONFIG_FILE);
            logger.info("Base configuration loaded from: {}", BASE_CONFIG_FILE);

            // Load environment-specific configuration
            String envConfigFile = String.format(ENV_CONFIG_FILE_TEMPLATE, environment.toLowerCase());
            loadPropertiesFile(envConfigFile);
            logger.info("Environment-specific configuration loaded from: {}", envConfigFile);
            logger.info("Framework initialized for environment: {}", environment);

        } catch (IOException e) {
            logger.error("Failed to load configuration files", e);
            throw new RuntimeException("Configuration loading failed. Ensure property files exist in resources/config/", e);
        }
    }

    /**
     * Loads properties from a specific file in the classpath.
     * 
     * @param fileName the path to the property file relative to resources
     * @throws IOException if file cannot be read
     */
    private void loadPropertiesFile(String fileName) throws IOException {
        try (InputStream inputStream = getClass().getClassLoader().getResourceAsStream(fileName)) {
            if (inputStream == null) {
                throw new IOException("Property file '" + fileName + "' not found in the classpath");
            }
            properties.load(inputStream);
        }
    }

    /**
     * Gets a property value by key.
     * 
     * @param key the property key
     * @return the property value, or null if not found
     */
    public String getProperty(String key) {
        return properties.getProperty(key);
    }

    /**
     * Gets a property value by key with a default value.
     * 
     * @param key the property key
     * @param defaultValue the default value if key is not found
     * @return the property value, or defaultValue if not found
     */
    public String getProperty(String key, String defaultValue) {
        return properties.getProperty(key, defaultValue);
    }
