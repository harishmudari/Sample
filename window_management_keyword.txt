## 📊 **Integration with Test Frameworks:**

### **TestNG Integration**
```java
public class WindowManagementTests extends BaseTest {
    
    private WindowManagementKeyword windowKeyword;
    
    @BeforeMethod
    public void setup() {
        windowKeyword = new WindowManagementKeyword(getDriver(), getScreenshotHelper());
    }
    
    @Test(description = "Test responsive design across different viewport sizes")
    public void testResponsiveDesign() {
        AdvancedWindowManagement advancedWindow = new AdvancedWindowManagement(windowKeyword);
        
        WindowSize[] testSizes = {
            WindowSize.MOBILE_PORTRAIT,
            WindowSize.TABLET_PORTRAIT, 
            WindowSize.DESKTOP_LARGE
        };
        
        for (WindowSize size : testSizes) {
            WindowResult result = advancedWindow.setWindowSize(size);
            Assert.assertTrue(result.isSuccess(), "Failed to set window size: " + size);
            
            // Run responsive tests at this size
            validateResponsiveLayout();
        }
    }
    
    @Test(description = "Test window state management")
    public void testWindowStateManagement() {
        // Test maximize
        WindowResult result = windowKeyword.maximizeWindow();
        Assert.assertTrue(result.isSuccess());
        Assert.assertTrue(windowKeyword.isWindowMaximized());
        
        // Test fullscreen
        result = windowKeyword.setFullscreen();
        Assert.assertTrue(result.isSuccess());
        
        // Exit fullscreen
        result = windowKeyword.exitFullscreen();
        Assert.assertTrue(result.isSuccess());
        Assert.assertFalse(windowKeyword.isWindowFullscreen());
    }
}
```
